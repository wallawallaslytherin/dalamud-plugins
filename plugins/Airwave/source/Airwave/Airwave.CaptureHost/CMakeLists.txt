cmake_minimum_required(VERSION 3.24)
project(AirwaveCapture LANGUAGES CXX)
add_executable(AirwaveCapture main.cpp)
set_target_properties(AirwaveCapture PROPERTIES OUTPUT_NAME "Airwave.Capture")
target_compile_features(AirwaveCapture PRIVATE cxx_std_20)
target_compile_definitions(AirwaveCapture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(AirwaveCapture PRIVATE /W4 /WX /EHsc /permissive-)
target_link_libraries(AirwaveCapture PRIVATE ole32 mmdevapi winmm)
set_property(TARGET AirwaveCapture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
